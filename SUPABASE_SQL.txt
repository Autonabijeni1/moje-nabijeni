-- Spusť v Supabase > SQL Editor > Run (jednou).
-- Každý přihlášený uživatel může číst a upravovat pouze svůj záznam.
create table if not exists public.moje_nabijeni_cloud (
  user_id uuid primary key references auth.users(id) on delete cascade,
  data jsonb not null,
  updated_at timestamptz not null default now()
);
alter table public.moje_nabijeni_cloud enable row level security;
revoke all on public.moje_nabijeni_cloud from anon;
grant select, insert, update, delete on public.moje_nabijeni_cloud to authenticated;
drop policy if exists "cloud_select_own" on public.moje_nabijeni_cloud;
drop policy if exists "cloud_insert_own" on public.moje_nabijeni_cloud;
drop policy if exists "cloud_update_own" on public.moje_nabijeni_cloud;
create policy "cloud_select_own" on public.moje_nabijeni_cloud for select to authenticated using ((select auth.uid()) = user_id);
create policy "cloud_insert_own" on public.moje_nabijeni_cloud for insert to authenticated with check ((select auth.uid()) = user_id);
create policy "cloud_update_own" on public.moje_nabijeni_cloud for update to authenticated using ((select auth.uid()) = user_id) with check ((select auth.uid()) = user_id);
